Início da função main
.text
.globl  main
main:

Prólogo:
/********************************************************/
/* mantenha este trecho aqui e nao mexa - prologo !!!   */
  pushq   %rbp
  movq    %rsp, %rbp
  subq    $16, %rsp
  movq    %rbx, -8(%rbp)  /* guarda rbx */
  movq    %r12, -16(%rbp)  /* guarda r12 */
/********************************************************/

equivalente a char *pc = S2, onde o ponteiro pc é o r12

movq  $S2, %r12  /* r12 = &S2 */

Confere se o valor apontado por (%r12) é zero, se for vai para 
o label L2, em seguida faz eax igual ao valor de (%r12)

L1:
  cmpb  $0, (%r12)  /* if (*pc == 0) ? */
  je  L2          /* goto L2 */

  movsbl  (%r12), %eax    /* eax = *r12 (estendendo o byte para 32 bits */

/*************************************************************/
/* este trecho imprime o valor de %eax (estraga %eax)  */
  movq    $Sf, %rdi    /* primeiro parametro (ponteiro)*/
  movl    %eax, %esi   /* segundo parametro  (inteiro) */
  movl  $0, %eax
  call  printf       /* chama a funcao da biblioteca */
/*************************************************************/

Incrementa o valor do ponteiro para andar pela lista
  addq  $1, %r12  /* r12 += 1; */
  jmp  L1         /* goto L1; */

Imprime '\n' e finaliza o programa
  L2:  
/*************************************************************/
/* este trecho imprime o \n (estraga %eax)                  */
  movq    $Sf2, %rdi    /* primeiro parametro (ponteiro)*/
  movl  $0, %eax
  call  printf       /* chama a funcao da biblioteca */
/*************************************************************/

/***************************************************************/
/* mantenha este trecho aqui e nao mexa - finalizacao!!!!      */
  movq  $0, %rax  /* rax = 0  (valor de retorno) */
  movq    -16(%rbp), %r12 /* recupera r12 */
  movq    -8(%rbp), %rbx  /* recupera rbx */
  leave
  ret      
/***************************************************************/

2-
A string é armazenada em .data como um array terminado em 0
Um ponteiro (%r12) callee-saved percorre a string byte a byte
usando o while (*pc)
Antes de imprimir cada caractere verificamos se o byte é '{' ou '}'
com cmpb e je

3-
O contador i é mantido em %ebx, registrador callee-saved
A cada iteração, i*i é calculado em %eax via imull e passado
como argumento para o printf como "%d\n"
%ebx é incrementado após a chamada de printf

4-
%rcx recebe o endereço de nums[0]
%edx recebe uma cópia do índice i e é multipllicado
por 4(sizeof(int)) via imull.
%rcx é somado com %rdx, resultando no endereço nums[i]
movl (%rcx), %edx lê o valor do elemento
O valor é acumulado em %ebx
